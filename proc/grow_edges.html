    pure subroutine grow_edges(w)
    type(walk_t), intent(inout) :: w    !! the working space
    integer, allocatable :: grown(:)
    integer :: n

    n = size(w%from)
    allocate (grown(2*n))
    grown(1:n) = w%from
    call move_alloc(grown, w%from)
    allocate (grown(2*n))
    grown(1:n) = w%to
    call move_alloc(grown, w%to)
end subroutine grow_edges
