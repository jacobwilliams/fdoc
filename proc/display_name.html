    pure function display_name(e) result(name)
    type(entity_t), intent(in) :: e     !! the entity
    character(len=display_length(e)) :: name
    integer :: at

    at = 0
    if (e%kind == E_FILE) at = last_separator(e%name)
    name = e%name(at + 1:)
end function display_name
