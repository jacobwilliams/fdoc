    pure function section_id(title) result(id)
    character(len=*), intent(in) :: title   !! the title
    character(len=len(title)) :: id
    integer :: k

    id = lowercase(title)
    do k = 1, len(id)
        if (id(k:k) == ' ') id(k:k) = '-'
    end do
end function section_id
