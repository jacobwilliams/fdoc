    pure function int_text(k) result(text)
    integer, intent(in) :: k    !! the number
    character(len=12) :: text
    character(len=12) :: digits
    integer :: at, rest

    at = len(digits) + 1
    rest = abs(k)
    do
        at = at - 1
        digits(at:at) = achar(iachar('0') + mod(rest, 10))
        rest = rest/10
        if (rest == 0) exit
    end do
    if (k < 0) then
        at = at - 1
        digits(at:at) = '-'
    end if
    text = digits(at:)
end function int_text
