    pure integer function count_blanks(s)
    character(len=*), intent(in) :: s   !! the string
    integer :: k
    count_blanks = 0
    do k = 1, len(s)
        if (s(k:k) == ' ' .or. s(k:k) == achar(9)) count_blanks = count_blanks + 1
    end do
end function count_blanks
