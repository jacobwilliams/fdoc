    elemental logical function starts_name(ch)
    character, intent(in) :: ch     !! the character
    starts_name = (ch >= 'a' .and. ch <= 'z') .or. (ch >= 'A' .and. ch <= 'Z') .or. ch == '_'
end function starts_name
