    subroutine number_cell(value, b)
    integer(int64), intent(in) :: value     !! the number
    type(buffer_t), intent(inout) :: b      !! the HTML

    call add(b, '<td class="num" data-sort="')
    call add_int(b, value)
    call add(b, '">')
    call grouped(value, b)
    call add(b, '</td>')
