    subroutine encoded(text, out)
    character(len=*), intent(in) :: text    !! the text
    type(buffer_t), intent(inout) :: out    !! the HTML
    integer :: k

    do k = 1, len(text)
        call add(out, '&#'//trim(int_text(iachar(text(k:k))))//';')
    end do
end subroutine encoded
