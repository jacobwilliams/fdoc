    pure integer function module_position(p, id)
    type(project_t), intent(in) :: p    !! the project
    integer, intent(in) :: id           !! the module
    integer :: k

    module_position = 0
    do k = 1, size(p%modules)
        if (p%modules(k) == id) then
            module_position = k
            return
        end if
    end do
end function module_position
