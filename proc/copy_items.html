    pure subroutine copy_items(from, to)
    type(value_t), intent(in) :: from(:)                !! the values
    type(value_t), allocatable, intent(out) :: to(:)    !! the copies
    integer :: k

    allocate (to(size(from)))
    do k = 1, size(from)
        call copy_value(to(k), from(k))
    end do
end subroutine copy_items
