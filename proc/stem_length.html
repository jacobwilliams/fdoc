    pure integer function stem_length(name)
    character(len=*), intent(in) :: name    !! the name of a Markdown file
    integer :: at

    stem_length = len(name) - 3
    at = index(name(:stem_length), '.', back=.true.)
    if (at > 1) stem_length = at - 1
end function stem_length
