    pure integer function last_child(t, parent)
    type(tree_t), intent(in) :: t   !! the conversion
    integer, intent(in) :: parent   !! the node

    last_child = 0
    if (size(t%nodes(parent)%kids) > 0) last_child = t%nodes(parent)%kids(size(t%nodes(parent)%kids))
