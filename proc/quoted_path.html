    function quoted_path(path) result(out)
    character(len=*), intent(in) :: path    !! the path
    character(len=len(path) + 2) :: out
    character(len=len(path)) :: native
    integer :: k

    native = path
    if (is_windows()) then
        do k = 1, len(native)
            if (native(k:k) == '/') native(k:k) = '\'
        end do
    end if
    out = quoted(native)
end function quoted_path
