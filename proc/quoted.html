    function quoted(text) result(out)
    character(len=*), intent(in) :: text    !! the text
    character(len=len(text) + 2) :: out

    if (is_windows()) then
        out = '"'//text//'"'
    else
        out = "'"//text//"'"
    end if
end function quoted
