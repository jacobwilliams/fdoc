    pure subroutine read_name(s)
    type(scan_t), intent(inout) :: s    !! the expression
    integer :: start

    start = s%pos
    do while (s%pos <= len(s%text))
        if (.not. in_name(s%text(s%pos:s%pos))) exit
        s%pos = s%pos + 1
    end do
    s%kind = T_NAME
    s%word = s%text(start:s%pos - 1)
end subroutine read_name
