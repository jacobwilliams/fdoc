    pure function trim_length(text) result(n)
    character(len=*), intent(in) :: text    !! the text
    integer :: n

    n = len(text)
    do while (n > 0)
        if (.not. is_blank(text(n:n))) exit
        n = n - 1
    end do
end function trim_length
