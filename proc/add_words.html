    subroutine add_words(words, b)
    type(string_t), intent(in) :: words(:)  !! the words
    type(buffer_t), intent(inout) :: b      !! the HTML
    integer :: k

    do k = 1, size(words)
        call add(b, '<li>')
        call html_escape(words(k)%s, b, .false.)
        call add(b, '</li>')
    end do
end subroutine add_words
