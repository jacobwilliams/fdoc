    subroutine argument_name(list, id, name)

    integer, intent(in) :: id                           !! the argument, or 0
    character(len=:), allocatable, intent(out) :: name  !! its name

    name = '*'
    if (id > 0) name = list%box(id)%e%name
end subroutine argument_name
