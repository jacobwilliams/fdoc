    pure logical function at_word(s, word)
    type(scan_t), intent(in) :: s           !! the expression
    character(len=*), intent(in) :: word    !! the word

    at_word = .false.
    if (s%kind /= T_OP .and. s%kind /= T_NAME) return
    if (len(s%word) /= len(word)) return
    at_word = s%word == word
end function at_word
